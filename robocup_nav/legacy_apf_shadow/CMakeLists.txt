cmake_minimum_required(VERSION 3.8)
project(robocup_legacy_apf_shadow)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(yaml-cpp REQUIRED)
set(LEGACY_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../src/uav_task")
# Recompile the actual existing controller, never the flight/servo communicator.
add_executable(legacy_apf_shadow node.cpp "${LEGACY_DIR}/src/velocity_apf_controller.cpp")
target_include_directories(legacy_apf_shadow PRIVATE "${LEGACY_DIR}/include")
ament_target_dependencies(legacy_apf_shadow rclcpp ament_index_cpp sensor_msgs std_msgs geometry_msgs nav_msgs)
target_link_libraries(legacy_apf_shadow yaml-cpp)
add_executable(path_guidance_test path_guidance_test.cpp)
target_compile_options(path_guidance_test PRIVATE -UNDEBUG)
